3. TABLE SPECIFICATIONS
================================================================================

--------------------------------------------------------------------------------
TABLE: departments
--------------------------------------------------------------------------------
Purpose: Stores information about 6 CCI departments
Primary Key: id (UUID)
Unique Keys: code (VARCHAR(10))

COLUMNS:
┌──────────────────┬──────────────┬──────────┬─────────────────────────────┐
│ Column           │ Type         │ Nullable │ Description                 │
├──────────────────┼──────────────┼──────────┼─────────────────────────────┤
│ id               │ UUID         │ NO       │ Primary key                 │
│ code             │ VARCHAR(10)  │ NO       │ Dept code: CS,SWE,IT,etc.   │
│ name             │ VARCHAR(100) │ NO       │ Full department name        │
│ description      │ TEXT         │ NO       │ Department overview         │
│ core_courses     │ TEXT[]       │ YES      │ Array of course names       │
│ career_paths     │ TEXT[]       │ YES      │ Career opportunities        │
│ industry_demand  │ VARCHAR(20)  │ YES      │ High/Medium/Low             │
│ color            │ VARCHAR(7)   │ YES      │ Hex color for UI            │
│ icon             │ VARCHAR(10)  │ YES      │ Icon identifier             │
│ created_at       │ TIMESTAMPTZ  │ NO       │ Creation timestamp          │
│ updated_at       │ TIMESTAMPTZ  │ NO       │ Last update timestamp       │
└──────────────────┴──────────────┴──────────┴─────────────────────────────┘

INDEXES:
- PRIMARY KEY: id
- UNIQUE: code
- INDEX: name
- GIN INDEX: core_courses (array search)

DATA:
6 departments: CS, SWE, IT, IS, ISC, STAT

--------------------------------------------------------------------------------
TABLE: questions
--------------------------------------------------------------------------------
Purpose: Assessment questions (20 questions)
Primary Key: id (UUID)
Unique Keys: question_order

COLUMNS:
┌──────────────────┬──────────────┬──────────┬─────────────────────────────┐
│ Column           │ Type         │ Nullable │ Description                 │
├──────────────────┼──────────────┼──────────┼─────────────────────────────┤
│ id               │ UUID         │ NO       │ Primary key                 │
│ text             │ TEXT         │ NO       │ Question text               │
│ category         │ VARCHAR(50)  │ NO       │ interests/skills/etc.       │
│ difficulty       │ VARCHAR(20)  │ YES      │ EASY/MEDIUM/HARD            │
│ question_order   │ INTEGER      │ NO       │ Display order (1-20)        │
│ is_active        │ BOOLEAN      │ NO       │ Active question?            │
│ created_at       │ TIMESTAMPTZ  │ NO       │ Creation timestamp          │
│ updated_at       │ TIMESTAMPTZ  │ NO       │ Last update timestamp       │

- category IN ('interests', 'skills', 'learning_style', 'career_goals', 'problem_solving')

- PRIMARY KEY: id
- UNIQUE: question_order

20 active questions

8. COMMON QUERIES
================================================================================

-- Get all departments with courses
SELECT code, name, core_courses 
FROM departments 
ORDER BY code;

-- Get active questions with options
SELECT q.*, qo.* 
FROM questions q
JOIN question_options qo ON q.id = qo.question_id
WHERE q.is_active = TRUE
ORDER BY q.question_order, qo.option_order;

-- Get assessment with responses
SELECT a.*, ar.*, q.text as question_text, qo.text as option_text
FROM assessments a
LEFT JOIN assessment_responses ar ON a.id = ar.assessment_id
LEFT JOIN questions q ON ar.question_id = q.id
LEFT JOIN question_options qo ON ar.option_id = qo.id
WHERE a.id = 'assessment-uuid-here';

-- Calculate department scores for an assessment
SELECT 
  SUM(qo.cs_score) as cs_total,
  SUM(qo.swe_score) as swe_total,
  SUM(qo.it_score) as it_total,
  SUM(qo.is_score) as is_total,
  SUM(qo.isc_score) as isc_total,
  SUM(qo.stat_score) as stat_total
FROM assessment_responses ar
JOIN question_options qo ON ar.option_id = qo.id
WHERE ar.assessment_id = 'assessment-uuid-here';

-- Get top recommendations for an assessment
SELECT r.*, d.name as department_name
FROM recommendations r
JOIN departments d ON r.department_id = d.code
WHERE r.assessment_id = 'assessment-uuid-here'
ORDER BY r.rank;

-- Get completion statistics
SELECT 
  COUNT(*) as total_assessments,
  COUNT(*) FILTER (WHERE completed_at IS NOT NULL) as completed,
  ROUND(AVG(EXTRACT(EPOCH FROM (completed_at - started_at))/60), 2) as avg_minutes
FROM assessments;
